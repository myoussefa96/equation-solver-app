# Storage + Migration Runtime Implementation Plan

> **For agentic workers:** Use superpowers:subagent-driven-development with fresh implementer/reviewer subagents per task.

**Goal:** Persist folders, entries, equations, and per-variable metadata in on-device SQLite via `sqflite`. Add a runtime "Import from old app" service that takes the importer's `ParsedFolder` list and writes it to the database. Tests use `sqflite_common_ffi` so they run on desktop CI.

**Architecture:** Per spec §4. Five tables (folders, entries, equations, variable_meta, solve_history) with FK CASCADE. Each table has one repository class. A single `Database` singleton handles open + migrations. The `ImportService` wraps `LegacyImporter` with the repos to persist a full ParsedFolder list atomically.

**Tech Stack:** Dart/Flutter 3.x, `sqflite ^2.4`, `sqflite_common_ffi ^2.3` (dev), `path ^1.9`. Engine layer untouched.

import 'dart:async';

import 'package:path/path.dart' as p;
import 'package:sqflite/sqflite.dart' as sf;
import 'package:sqflite_common/sqlite_api.dart';

class AppDatabase {
  static const int schemaVersion = 1;

  final Database raw;
  AppDatabase._(this.raw);

  /// Open the on-device app database. Used by the running Flutter app.
  static Future<AppDatabase> openApp() async {
    final dir = await sf.getDatabasesPath();
    final path = p.join(dir, 'equation_solver.db');
    final db = await sf.openDatabase(
      path,
      version: schemaVersion,
      onConfigure: _onConfigure,
      onCreate: _onCreate,
    );
    return AppDatabase._(db);
  }

  /// Open an in-memory database — used by tests.
  static Future<AppDatabase> openInMemory({
    required DatabaseFactory databaseFactory,
  }) async {
    final db = await databaseFactory.openDatabase(
      inMemoryDatabasePath,
      options: OpenDatabaseOptions(
        version: schemaVersion,
        onConfigure: _onConfigure,
        onCreate: _onCreate,
      ),
    );
    return AppDatabase._(db);
  }

  static Future<void> _onConfigure(Database db) async {
    await db.execute('PRAGMA foreign_keys = ON');
  }

  static Future<void> _onCreate(Database db, int version) async {
    for (final stmt in _schema) {
      await db.execute(stmt);
    }
  }

  Future<void> close() => raw.close();

  Future<T> transaction<T>(Future<T> Function(Transaction txn) action) =>
      raw.transaction(action);
}

const List<String> _schema = [

  ''',

    description TEXT,

import 'package:equation_solver_app/storage/database.dart';

class FolderRow {
  final int id;
  final String name;
  final int sortOrder;
  final DateTime createdAt;
  final DateTime updatedAt;

  const FolderRow({
    required this.id,
    required this.name,
    required this.sortOrder,
    required this.createdAt,
    required this.updatedAt,
  });

  factory FolderRow.fromMap(Map<String, Object?> m) => FolderRow(
        id: m['id']! as int,
        name: m['name']! as String,
        sortOrder: m['sort_order']! as int,
        createdAt: DateTime.parse(m['created_at']! as String),
        updatedAt: DateTime.parse(m['updated_at']! as String),
      );
}

class FoldersRepo {
  final AppDatabase _db;
  const FoldersRepo(this._db);

  Future<int> create({required String name, int sortOrder = 0}) async {
    final now = DateTime.now().toUtc().toIso8601String();
    return _db.raw.insert('folders', {
      'name': name,
      'sort_order': sortOrder,
      'created_at': now,
      'updated_at': now,
    });
  }

  Future<void> rename(int id, String newName) async {
    final now = DateTime.now().toUtc().toIso8601String();
    await _db.raw.update(
      'folders',
      {'name': newName, 'updated_at': now},
      where: 'id = ?',
      whereArgs: [id],
    );
  }

  Future<void> delete(int id) =>
      _db.raw.delete('folders', where: 'id = ?', whereArgs: [id]);

  Future<List<FolderRow>> listAll() async {
    final rows = await _db.raw.query('folders', orderBy: 'sort_order ASC, id ASC');
    return rows.map(FolderRow.fromMap).toList();
  }

  Future<FolderRow?> getById(int id) async {
    final rows = await _db.raw.query('folders', where: 'id = ?', whereArgs: [id]);
    if (rows.isEmpty) return null;
    return FolderRow.fromMap(rows.first);
  }
}

import 'dart:convert';

import 'package:equation_solver_app/engine/ast.dart';
import 'package:equation_solver_app/engine/ast_json.dart';
import 'package:equation_solver_app/storage/database.dart';

class EquationDraft {
  final int entryId;
  final String outputVar;
  final AstNode ast;
  final int sortOrder;

  const EquationDraft({
    required this.entryId,
    required this.outputVar,
    required this.ast,
    required this.sortOrder,
  });
}

class EquationRow {
  final int id;
  final int entryId;
  final String outputVar;
  final AstNode ast;
  final int sortOrder;

  const EquationRow({
    required this.id,
    required this.entryId,
    required this.outputVar,
    required this.ast,
    required this.sortOrder,
  });
}

class EquationsRepo {
  final AppDatabase _db;
  const EquationsRepo(this._db);

  Future<int> put(EquationDraft draft) async {
    return _db.raw.insert('equations', {
      'entry_id': draft.entryId,
      'output_var': draft.outputVar,
      'ast_json': jsonEncode(astToJson(draft.ast)),
      'sort_order': draft.sortOrder,
    });
  }

  /// Atomically replace all equations for an entry.
  Future<void> replaceAll(int entryId, List<EquationDraft> drafts) async {
    await _db.transaction((txn) async {
      await txn.delete('equations', where: 'entry_id = ?', whereArgs: [entryId]);
      for (final d in drafts) {
        await txn.insert('equations', {
          'entry_id': d.entryId,
          'output_var': d.outputVar,
          'ast_json': jsonEncode(astToJson(d.ast)),
          'sort_order': d.sortOrder,
        });
      }
    });
  }

  Future<List<EquationRow>> listByEntry(int entryId) async {
    final rows = await _db.raw.query(
      'equations',
      where: 'entry_id = ?',
      whereArgs: [entryId],
      orderBy: 'sort_order ASC, id ASC',
    );
    return rows
        .map((m) => EquationRow(
              id: m['id']! as int,
              entryId: m['entry_id']! as int,
              outputVar: m['output_var']! as String,
              ast: astFromJson(
                jsonDecode(m['ast_json']! as String) as Map<String, Object?>,
              ),
              sortOrder: m['sort_order']! as int,
            ))
        .toList();
  }

  Future<void> deleteById(int id) =>
      _db.raw.delete('equations', where: 'id = ?', whereArgs: [id]);
}

    required this.role,
    this.isFinal = false,
    this.lastValue,
  });
}

class VariableMetaRow {
  final int id;
  final int entryId;
  final String identifier;
  final String displayLabel;
  final String? unit;
  final VarRole role;
  final bool isFinal;
  final double? lastValue;

  const VariableMetaRow({
    required this.id,
    required this.entryId,
    required this.identifier,
    required this.displayLabel,
    required this.unit,
    required this.role,
    required this.isFinal,
    required this.lastValue,
  });
}

class VariableMetaRepo {
  final AppDatabase _db;
  const VariableMetaRepo(this._db);

  static String _roleText(VarRole r) => switch (r) {
        VarRole.input => 'input',
        VarRole.intermediate => 'intermediate',
        VarRole.output => 'output',
      };

  static VarRole _roleFromText(String s) => switch (s) {
        'input' => VarRole.input,
        'intermediate' => VarRole.intermediate,
        'output' => VarRole.output,
        _ => throw FormatException('Unknown role: $s'),
      };

  Future<int> put(VariableMetaDraft d) {
    return _db.raw.insert('variable_meta', {
      'entry_id': d.entryId,
      'identifier': d.identifier,
      'display_label': d.displayLabel,
      'unit': d.unit,
      'role': _roleText(d.role),
      'is_final': d.isFinal ? 1 : 0,
      'last_value': d.lastValue,
    });
  }

  Future<void> replaceAll(int entryId, List<VariableMetaDraft> drafts) async {
    await _db.transaction((txn) async {
      await txn.delete('variable_meta',
          where: 'entry_id = ?', whereArgs: [entryId]);
      for (final d in drafts) {
        await txn.insert('variable_meta', {
          'entry_id': d.entryId,
          'identifier': d.identifier,
          'display_label': d.displayLabel,
          'unit': d.unit,
          'role': _roleText(d.role),
          'is_final': d.isFinal ? 1 : 0,
          'last_value': d.lastValue,
        });
      }
    });
  }

  Future<void> updateLastValue(int entryId, String identifier, double value) {
    return _db.raw.update(
      'variable_meta',
      {'last_value': value},
      where: 'entry_id = ? AND identifier = ?',
      whereArgs: [entryId, identifier],
    );
  }

  Future<List<VariableMetaRow>> listByEntry(int entryId) async {
    final rows = await _db.raw.query(
      'variable_meta',
      where: 'entry_id = ?',
      whereArgs: [entryId],
      orderBy: 'id ASC',
    );
    return rows
        .map((m) => VariableMetaRow(
              id: m['id']! as int,
              entryId: m['entry_id']! as int,
              identifier: m['identifier']! as String,
              displayLabel: m['display_label']! as String,
              unit: m['unit'] as String?,
              role: _roleFromText(m['role']! as String),
              isFinal: (m['is_final']! as int) != 0,
              lastValue: m['last_value'] as double?,
            ))
        .toList();
  }
}

import 'package:equation_solver_app/engine/parsed_entry.dart';
import 'package:equation_solver_app/storage/database.dart';
import 'package:equation_solver_app/storage/entries_repo.dart';
import 'package:equation_solver_app/storage/equations_repo.dart';
import 'package:equation_solver_app/storage/folders_repo.dart';
import 'package:equation_solver_app/storage/variable_meta_repo.dart';

class ImportService {
  final FoldersRepo folders;
  final EntriesRepo entries;
  final EquationsRepo equations;
  final VariableMetaRepo variables;

  const ImportService({
    required this.folders,
    required this.entries,
    required this.equations,
    required this.variables,
  });

  /// Write the imported folders into the DB. Each folder is created with the
  /// provided name; collisions are handled by appending " (imported)" if
  /// [replaceExisting] is false. If [replaceExisting] is true, ALL existing
  /// folders are deleted first (which cascades through entries / equations /
  /// variable_meta).
  Future<void> importParsedFolders(
    List<ParsedFolder> source, {
    bool replaceExisting = false,
  }) async {
    if (replaceExisting) {
      for (final f in await folders.listAll()) {
        await folders.delete(f.id);
      }
    }

    final existingNames =
        (await folders.listAll()).map((f) => f.name).toSet();

    for (var i = 0; i < source.length; i++) {
      final pf = source[i];
      var name = pf.name;
      while (existingNames.contains(name)) {
        name = '${pf.name} (imported)';
        if (existingNames.contains(name)) {
          // append index suffix to avoid infinite loop
          name = '${pf.name} (imported ${DateTime.now().microsecondsSinceEpoch})';
        }
      }
      existingNames.add(name);
      final folderId = await folders.create(name: name, sortOrder: i);

      for (var j = 0; j < pf.entries.length; j++) {
        final pe = pf.entries[j];
        final entryId = await entries.create(
          folderId: folderId,
          name: pe.name,
          description: pe.description,
          sortOrder: j,
        );

        await equations.replaceAll(entryId, [
          for (var k = 0; k < pe.equations.length; k++)
            EquationDraft(
              entryId: entryId,
              outputVar: pe.equations[k].outputVar,
              ast: pe.equations[k].ast,
              sortOrder: k,
            ),
        ]);

        await variables.replaceAll(entryId, [
          for (final v in pe.variables)
            VariableMetaDraft(
              entryId: entryId,
              identifier: v.identifier,
              displayLabel: v.displayLabel,
              unit: v.unit,
              role: v.role,
              isFinal: v.isFinal,
            ),
        ]);
      }
    }
  }
}
